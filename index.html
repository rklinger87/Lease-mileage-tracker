  <script>
    const STORAGE_KEY = "lease-mileage-tracker-html-v2";

    const defaultData = {
      setup: {
        startingOdometer: 0,
        leaseStart: "",
        leaseEnd: "",
        totalAllowedMiles: 45000
      },
      newReading: {
        date: "",
        odometer: ""
      },
      readings: []
    };

    let appData = loadData();
    let chartInstance = null;

    const els = {
      startingOdometer: document.getElementById("startingOdometer"),
      leaseStart: document.getElementById("leaseStart"),
      leaseEnd: document.getElementById("leaseEnd"),
      totalAllowedMiles: document.getElementById("totalAllowedMiles"),
      readingDate: document.getElementById("readingDate"),
      readingOdometer: document.getElementById("readingOdometer"),
      saveReadingButton: document.getElementById("saveReadingButton"),
      clearAllButton: document.getElementById("clearAllButton"),
      savedReadings: document.getElementById("savedReadings"),
      asOfBanner: document.getElementById("asOfBanner"),
      totalMilesValue: document.getElementById("totalMilesValue"),
      milesRemainingValue: document.getElementById("milesRemainingValue"),
      avgMilesValue: document.getElementById("avgMilesValue"),
      avgMilesSubtitle: document.getElementById("avgMilesSubtitle"),
      paceDifferenceValue: document.getElementById("paceDifferenceValue"),
      neededDailyValue: document.getElementById("neededDailyValue"),
      neededDailySubtitle: document.getElementById("neededDailySubtitle"),
      tilePaceDifference: document.getElementById("tilePaceDifference"),
      quickSummary: document.getElementById("quickSummary")
    };

    const tabButtons = document.querySelectorAll(".tab-button");
    const sections = document.querySelectorAll(".section");

    tabButtons.forEach((button) => {
      button.addEventListener("click", () => {
        const target = button.dataset.tab;
        tabButtons.forEach((b) => b.classList.remove("active"));
        sections.forEach((s) => s.classList.remove("active"));
        button.classList.add("active");
        document.getElementById(target).classList.add("active");
      });
    });

    els.startingOdometer.addEventListener("input", handleSetupInput);
    els.leaseStart.addEventListener("input", handleSetupInput);
    els.leaseEnd.addEventListener("input", handleSetupInput);
    els.totalAllowedMiles.addEventListener("input", handleSetupInput);
    els.readingDate.addEventListener("input", handleNewReadingInput);
    els.readingOdometer.addEventListener("input", handleNewReadingInput);
    els.saveReadingButton.addEventListener("click", saveReading);
    els.clearAllButton.addEventListener("click", clearAllData);

    fillInputs();
    renderAll();

    function cloneDefaultData() {
      return JSON.parse(JSON.stringify(defaultData));
    }

    function loadData() {
      try {
        const saved = localStorage.getItem(STORAGE_KEY);
        return saved ? JSON.parse(saved) : cloneDefaultData();
      } catch (error) {
        return cloneDefaultData();
      }
    }

    function saveData() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(appData));
    }

    function fillInputs() {
      els.startingOdometer.value = appData.setup.startingOdometer || "";
      els.leaseStart.value = appData.setup.leaseStart || "";
      els.leaseEnd.value = appData.setup.leaseEnd || "";
      els.totalAllowedMiles.value = appData.setup.totalAllowedMiles || "";
      els.readingDate.value = appData.newReading.date || "";
      els.readingOdometer.value = appData.newReading.odometer || "";
    }

    function handleSetupInput() {
      appData.setup.startingOdometer = Number(els.startingOdometer.value || 0);
      appData.setup.leaseStart = els.leaseStart.value;
      appData.setup.leaseEnd = els.leaseEnd.value;
      appData.setup.totalAllowedMiles = Number(els.totalAllowedMiles.value || 0);
      saveData();
      renderAll();
    }

    function handleNewReadingInput() {
      appData.newReading.date = els.readingDate.value;
      appData.newReading.odometer = els.readingOdometer.value;
      saveData();
    }

    function saveReading() {
      const date = appData.newReading.date;
      const odometer = Number(appData.newReading.odometer);

      if (!date || !Number.isFinite(odometer) || odometer < 0) {
        alert("Please enter a valid reading date and odometer value.");
        return;
      }

      const existing = appData.readings.find((r) => r.date === date);
      if (existing) {
        existing.odometer = odometer;
      } else {
        appData.readings.push({ id: Date.now(), date, odometer });
      }

      appData.readings.sort((a, b) => new Date(a.date) - new Date(b.date));
      saveData();
      renderAll();
      alert("Reading saved.");
    }

    function clearAllData() {
      const confirmed = confirm("Clear all saved lease settings and mileage readings on this device?");
      if (!confirmed) return;
      localStorage.removeItem(STORAGE_KEY);
      appData = cloneDefaultData();
      fillInputs();
      renderAll();
      alert("All data cleared.");
    }

    function deleteReading(id) {
      appData.readings = appData.readings.filter((reading) => reading.id !== id);
      saveData();
      renderAll();
    }

    function getSortedReadings() {
      return [...appData.readings].sort((a, b) => new Date(a.date) - new Date(b.date));
    }

    function daysBetween(start, end) {
      if (!start || !end) return 0;
      const s = new Date(start + "T00:00:00");
      const e = new Date(end + "T00:00:00");
      const ms = e.getTime() - s.getTime();
      return Math.max(0, Math.round(ms / 86400000));
    }

    function formatNumber(value, digits = 0) {
      if (!Number.isFinite(value)) return "—";
      return Number(value).toLocaleString(undefined, {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
      });
    }

    function formatDate(dateString) {
      if (!dateString) return "—";
      const date = new Date(dateString + "T00:00:00");
      return date.toLocaleDateString(undefined, {
        month: "short",
        day: "numeric",
        year: "numeric"
      });
    }

    function paceLabel(value) {
      if (!Number.isFinite(value)) return "—";
      const rounded = Math.round(value);
      if (rounded === 0) return "On pace";
      return `${Math.abs(rounded).toLocaleString()} miles ${rounded > 0 ? "ahead of schedule" : "behind schedule"}`;
    }

    function calculateMetrics() {
      const sortedReadings = getSortedReadings();
      const lastReading = sortedReadings.length ? sortedReadings[sortedReadings.length - 1] : null;
      const totalLeaseDays = daysBetween(appData.setup.leaseStart, appData.setup.leaseEnd);
      const startingOdometer = Number(appData.setup.startingOdometer);
      const totalAllowedMiles = Number(appData.setup.totalAllowedMiles);

      if (!lastReading) {
        return {
          lastReading: null,
          totalLeaseDays,
          elapsedDays: 0,
          remainingDays: 0,
          totalMilesTraveled: 0,
          milesRemaining: totalAllowedMiles,
          avgMilesPerDay: 0,
          expectedMilesByNow: 0,
          paceVariance: 0,
          neededDailyAverage: 0
        };
      }

      const elapsedDays = daysBetween(appData.setup.leaseStart, lastReading.date);
      const remainingDays = daysBetween(lastReading.date, appData.setup.leaseEnd);
      const totalMilesTraveled = Number(lastReading.odometer) - startingOdometer;
      const milesRemaining = totalAllowedMiles - totalMilesTraveled;
      const avgMilesPerDay = elapsedDays > 0 ? totalMilesTraveled / elapsedDays : 0;
      const expectedMilesByNow = totalLeaseDays > 0 ? totalAllowedMiles * (elapsedDays / totalLeaseDays) : 0;
      const paceVariance = totalMilesTraveled - expectedMilesByNow;
      const neededDailyAverage = remainingDays > 0 ? milesRemaining / remainingDays : 0;

      return {
        lastReading,
        totalLeaseDays,
        elapsedDays,
        remainingDays,
        totalMilesTraveled,
        milesRemaining,
        avgMilesPerDay,
        expectedMilesByNow,
        paceVariance,
        neededDailyAverage
      };
    }

    function buildChartData() {
      const sortedReadings = getSortedReadings();
      const totalLeaseDays = daysBetween(appData.setup.leaseStart, appData.setup.leaseEnd);
      const allowedMiles = Number(appData.setup.totalAllowedMiles);
      const startingOdometer = Number(appData.setup.startingOdometer);

      if (!appData.setup.leaseStart || !appData.setup.leaseEnd || totalLeaseDays <= 0) {
        return null;
      }

      const actualPoints = [
        { x: 0, y: 0 },
        ...sortedReadings.map((reading) => ({
          x: daysBetween(appData.setup.leaseStart, reading.date),
          y: Number(reading.odometer) - startingOdometer
        }))
      ].sort((a, b) => a.x - b.x);

      const goalPoints = [
        { x: 0, y: 0 },
        { x: totalLeaseDays, y: allowedMiles }
      ];

      return { actualPoints, goalPoints, totalLeaseDays };
    }

    function renderSavedReadings() {
      const sortedReadings = getSortedReadings();
      if (!sortedReadings.length) {
        els.savedReadings.innerHTML = '<div style="font-size:14px;color:#475569;">No readings saved yet.</div>';
        return;
      }

      els.savedReadings.innerHTML = sortedReadings.map((reading) => `
        <div class="reading-row">
          <div>
            <strong>${formatDate(reading.date)}</strong>
            <span>${formatNumber(reading.odometer)} miles</span>
          </div>
          <button class="delete-button" onclick="deleteReading(${reading.id})" aria-label="Delete reading">🗑️</button>
        </div>
      `).join("");
    }

    function renderMetrics(metrics) {
      const asOf = metrics.lastReading ? formatDate(metrics.lastReading.date) : "—";
      els.asOfBanner.textContent = `Metrics are based on mileage as of ${asOf}`;
      els.totalMilesValue.textContent = formatNumber(metrics.totalMilesTraveled);
      els.milesRemainingValue.textContent = formatNumber(metrics.milesRemaining);
      els.avgMilesValue.textContent = formatNumber(metrics.avgMilesPerDay, 1);
      els.avgMilesSubtitle.textContent = `${formatNumber(metrics.elapsedDays)} days elapsed as of ${asOf}`;
      els.paceDifferenceValue.textContent = paceLabel(metrics.paceVariance);
      els.neededDailyValue.textContent = formatNumber(metrics.neededDailyAverage, 1);
      els.neededDailySubtitle.textContent = `${formatNumber(metrics.remainingDays)} days remaining after ${asOf}`;

      els.tilePaceDifference.classList.remove("red", "green");
      if (metrics.paceVariance > 0) {
        els.tilePaceDifference.classList.add("red");
      } else if (metrics.paceVariance < 0) {
        els.tilePaceDifference.classList.add("green");
      }

      els.quickSummary.innerHTML = `
        <div>Expected miles by ${asOf}: <strong>${formatNumber(metrics.expectedMilesByNow)}</strong></div>
        <div>Actual miles by ${asOf}: <strong>${formatNumber(metrics.totalMilesTraveled)}</strong></div>
        <div>You will need to average <strong>${formatNumber(metrics.neededDailyAverage, 1)}</strong> miles for the next <strong>${formatNumber(metrics.remainingDays)}</strong> days to end with <strong>${formatNumber(appData.setup.totalAllowedMiles)}</strong> miles.</div>
      `;
    }

    function renderChart(chartData) {
      const ctx = document.getElementById("mileageChart").getContext("2d");
      if (chartInstance) {
        chartInstance.destroy();
      }

      if (!chartData) return;

      const leaseStartDate = appData.setup.leaseStart;
      const totalLeaseDays = chartData.totalLeaseDays;

      chartInstance = new Chart(ctx, {
        type: "line",
        data: {
          datasets: [
            {
              label: "Goal pace",
              data: chartData.goalPoints,
              parsing: false,
              borderColor: "#2563eb",
              backgroundColor: "#2563eb",
              borderWidth: 3,
              pointRadius: 0,
              tension: 0
            },
            {
              label: "Actual readings",
              data: chartData.actualPoints,
              parsing: false,
              borderColor: "#dc2626",
              backgroundColor: "#dc2626",
              borderWidth: 3,
              pointRadius: 4,
              pointHoverRadius: 5,
              tension: 0.2
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          interaction: { mode: "nearest", intersect: false },
          plugins: {
            legend: { display: true },
            tooltip: {
              callbacks: {
                title: function(items) {
                  if (!items.length) return "";
                  const dayOffset = items[0].raw.x;
                  const date = new Date(leaseStartDate + "T00:00:00");
                  date.setDate(date.getDate() + dayOffset);
                  return date.toLocaleDateString(undefined, {
                    month: "short",
                    day: "numeric",
                    year: "numeric"
                  });
                },
                label: function(context) {
                  return `${context.dataset.label}: ${formatNumber(context.raw.y)} miles`;
                }
              }
            }
          },
          scales: {
            x: {
              type: "linear",
              min: 0,
              max: totalLeaseDays,
              ticks: {
                stepSize: Math.max(1, Math.round(totalLeaseDays / 6)),
                callback: function(value) {
                  const date = new Date(leaseStartDate + "T00:00:00");
                  date.setDate(date.getDate() + value);
                  return date.toLocaleDateString(undefined, {
                    month: "short",
                    year: "2-digit"
                  });
                },
                maxRotation: 0,
                autoSkip: false
              },
              title: { display: true, text: "Lease timeline" }
            },
            y: {
              beginAtZero: true,
              title: { display: true, text: "Miles used" }
            }
          }
        }
      });
    }

    function renderAll() {
      renderSavedReadings();
      const metrics = calculateMetrics();
      renderMetrics(metrics);
      const chartData = buildChartData();
      renderChart(chartData);
      saveData();
    }

    window.deleteReading = deleteReading;
  </script>
